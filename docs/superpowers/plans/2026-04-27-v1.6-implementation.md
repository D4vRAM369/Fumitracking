## Task 1: Fix Midnight — Handler en MainActivity

**Archivos:**
- Modificar: `app/src/main/java/com/d4vram/cbdcounter/MainActivity.kt`

**Problema:** `ACTION_DATE_CHANGED` no es confiable cuando la app está en foreground. La solución robusta es un `Handler.postDelayed()` que se dispara exactamente a las 00:00:01 del día siguiente.

- [ ] **Step 1: Añadir campo handler y runnable en MainActivity**

Busca los campos de clase en `MainActivity.kt` (zona de `private val dateChangeReceiver`). Añade debajo:

```kotlin
private val handler = Handler(Looper.getMainLooper())
private val midnightReloadRunnable = Runnable {
    loadTodayData()
    updateDisplay(animate = false)
    loadAllHistoryData()
    updateHistoryView()
    updateStats()
    historyAdapter.refresh()
    scheduleMidnightReload()
}
```

- [ ] **Step 2: Añadir función scheduleMidnightReload()**

Añade esta función privada en MainActivity (después de `onPause()`):

```kotlin
private fun scheduleMidnightReload() {
    handler.removeCallbacks(midnightReloadRunnable)
    val midnight = Calendar.getInstance().apply {
        add(Calendar.DAY_OF_YEAR, 1)
        set(Calendar.HOUR_OF_DAY, 0)
        set(Calendar.MINUTE, 0)
        set(Calendar.SECOND, 1)
        set(Calendar.MILLISECOND, 0)
    }
    val delay = midnight.timeInMillis - System.currentTimeMillis()
    handler.postDelayed(midnightReloadRunnable, delay)
}
```

- [ ] **Step 3: Llamar scheduleMidnightReload() en onResume()**

En `onResume()` (línea ~163), añade al final:

- [ ] **Step 4: Cancelar handler en onPause()**

En `onPause()` (línea ~178), añade al inicio del try block:

```kotlin
object HeatmapUtils {

    fun colorForCount(count: Int): Int = when {
        count == 0  -> 0xFF1a472a.toInt()  // verde oscuro
        count <= 2  -> 0xFF2d6a4f.toInt()  // verde
        count <= 4  -> 0xFF52b788.toInt()  // verde claro
        count == 5  -> 0xFFf4a261.toInt()  // naranja
        count <= 7  -> 0xFFe76f51.toInt()  // naranja-rojo
        count <= 10 -> 0xFFc1121f.toInt()  // rojo
        else        -> 0xFF6d1a36.toInt()  // rojo oscuro
    }
}
```

**Archivos:**

- Modificar: `app/src/main/AndroidManifest.xml`

- [ ] **Step 1: Crear GoalsActivity.kt**

```kotlin
package com.d4vram.cbdcounter

import android.Manifest
import android.app.NotificationChannel
import android.app.NotificationManager
import android.content.Context
import android.content.pm.PackageManager
import android.os.Build
import android.os.Bundle
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import androidx.core.app.ActivityCompat
import androidx.core.app.NotificationCompat
import androidx.core.app.NotificationManagerCompat
import com.google.android.material.button.MaterialButton
import com.google.android.material.switchmaterial.SwitchMaterial
import java.text.SimpleDateFormat
import java.util.*

class GoalsActivity : AppCompatActivity() {

    companion object {
        const val PREFS_NAME = "goals_prefs"
        const val KEY_CBD_MAX = "goal_cbd_max"
        const val KEY_THC_MAX = "goal_thc_max"
        const val KEY_NOTIFICATIONS = "goal_notifications_enabled"
        const val CHANNEL_ID = "goals_channel"
        const val NOTIF_ID = 1001
    }

    private val prefs by lazy { getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE) }

    private var cbdGoal = 0
    private var thcGoal = 0

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_goals)
        supportActionBar?.title = "Metas Diarias"
        supportActionBar?.setDisplayHomeAsUpEnabled(true)

        createNotificationChannel()
        loadGoals()
        setupButtons()
        updateStreakDisplay()
    }

    override fun onSupportNavigateUp(): Boolean {
        finish()
        return true
    }

    private fun loadGoals() {
        cbdGoal = prefs.getInt(KEY_CBD_MAX, 0)
        thcGoal = prefs.getInt(KEY_THC_MAX, 0)
        val notifEnabled = prefs.getBoolean(KEY_NOTIFICATIONS, false)

        findViewById<TextView>(R.id.tvCbdGoal).text = cbdGoal.toString()
        findViewById<TextView>(R.id.tvThcGoal).text = thcGoal.toString()
        findViewById<SwitchMaterial>(R.id.switchGoalNotification).isChecked = notifEnabled
    }

    private fun setupButtons() {
        val tvCbd = findViewById<TextView>(R.id.tvCbdGoal)
        val tvThc = findViewById<TextView>(R.id.tvThcGoal)
        val switchNotif = findViewById<SwitchMaterial>(R.id.switchGoalNotification)

        findViewById<MaterialButton>(R.id.btnCbdMinus).setOnClickListener {
            if (cbdGoal > 0) cbdGoal--
            tvCbd.text = cbdGoal.toString()
            saveGoals()
        }
        findViewById<MaterialButton>(R.id.btnCbdPlus).setOnClickListener {
            cbdGoal++
            tvCbd.text = cbdGoal.toString()
            saveGoals()
        }
        findViewById<MaterialButton>(R.id.btnThcMinus).setOnClickListener {
            if (thcGoal > 0) thcGoal--
            tvThc.text = thcGoal.toString()
            saveGoals()
        }
        findViewById<MaterialButton>(R.id.btnThcPlus).setOnClickListener {
            thcGoal++
            tvThc.text = thcGoal.toString()
            saveGoals()
        }
        switchNotif.setOnCheckedChangeListener { _, checked ->
            if (checked && Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
                if (checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED) {
                    ActivityCompat.requestPermissions(this, arrayOf(Manifest.permission.POST_NOTIFICATIONS), 100)
                }
            }
            prefs.edit().putBoolean(KEY_NOTIFICATIONS, checked).apply()
        }
    }

    private fun saveGoals() {
        prefs.edit()
            .putInt(KEY_CBD_MAX, cbdGoal)
            .putInt(KEY_THC_MAX, thcGoal)
            .apply()
        updateStreakDisplay()
    }

    private fun updateStreakDisplay() {
        val streak = calculateStreak()
        val label = if (streak == 1) "día" else "días"
        findViewById<TextView>(R.id.tvStreak).text = "🔥 $streak $label"
    }

    private fun calculateStreak(): Int {
        if (cbdGoal == 0 && thcGoal == 0) return 0
        val fmt = SimpleDateFormat("dd/MM/yyyy", Locale.getDefault())
        val cal = Calendar.getInstance()
        cal.add(Calendar.DAY_OF_YEAR, -1)
        var streak = 0
        repeat(365) {
            val dateKey = fmt.format(cal.time)
            val cbd = Prefs.getCbdCount(this, dateKey)
            val thc = Prefs.getThcCount(this, dateKey)
            val cbdOk = cbdGoal == 0 || cbd <= cbdGoal
            val thcOk = thcGoal == 0 || thc <= thcGoal
            if (cbdOk && thcOk) {
                streak++
                cal.add(Calendar.DAY_OF_YEAR, -1)
            } else {
                return streak
            }
        }
        return streak
    }

    private fun createNotificationChannel() {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            val channel = NotificationChannel(
                CHANNEL_ID,
                "Metas de consumo",
                NotificationManager.IMPORTANCE_DEFAULT
            ).apply { description = "Alertas al superar meta diaria" }
            getSystemService(NotificationManager::class.java).createNotificationChannel(channel)
        }
    }

    companion object {
        fun notifyIfGoalExceeded(context: Context, cbdCount: Int, thcCount: Int) {
            val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
            if (!prefs.getBoolean(KEY_NOTIFICATIONS, false)) return

            val cbdGoal = prefs.getInt(KEY_CBD_MAX, 0)
            val thcGoal = prefs.getInt(KEY_THC_MAX, 0)
            if (cbdGoal == 0 && thcGoal == 0) return

            val today = SimpleDateFormat("dd/MM/yyyy", Locale.getDefault()).format(Date())
            val notifKey = "notified_$today"
            if (prefs.getBoolean(notifKey, false)) return

            val cbdExceeded = cbdGoal > 0 && cbdCount > cbdGoal
            val thcExceeded = thcGoal > 0 && thcCount > thcGoal
            if (!cbdExceeded && !thcExceeded) return

            val msg = buildString {
                if (cbdExceeded) append("CBD: $cbdCount/$cbdGoal ")
                if (thcExceeded) append("THC: $thcCount/$thcGoal")
            }

            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU &&
                context.checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
            ) return

            val notif = NotificationCompat.Builder(context, CHANNEL_ID)
                .setSmallIcon(android.R.drawable.ic_dialog_info)
                .setContentTitle("Meta superada")
                .setContentText(msg.trim())
                .setPriority(NotificationCompat.PRIORITY_DEFAULT)
                .setAutoCancel(true)
                .build()

            NotificationManagerCompat.from(context).notify(NOTIF_ID, notif)
            prefs.edit().putBoolean(notifKey, true).apply()
        }
    }
}
```

## Task 10: InsightsActivity — Kotlin

**Archivos:**

- [ ] **Step 1: Crear InsightsActivity.kt**

```kotlin
package com.d4vram.cbdcounter

import android.graphics.Color
import android.os.Bundle
import android.view.Gravity
import android.view.View
import android.widget.LinearLayout
import android.widget.TextView
import androidx.appcompat.app.AppCompatActivity
import androidx.cardview.widget.CardView
import com.google.android.material.button.MaterialButton
import java.text.SimpleDateFormat
import java.util.*
import kotlin.math.abs
import kotlin.math.roundToInt

class InsightsActivity : AppCompatActivity() {

    private val fmt = SimpleDateFormat("dd/MM/yyyy", Locale.getDefault())
    private var weekOffset = 0  // 0 = esta semana, -1 = semana anterior, etc.

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_insights)
        supportActionBar?.title = "Insights"
        supportActionBar?.setDisplayHomeAsUpEnabled(true)

        findViewById<MaterialButton>(R.id.btnPrevWeek).setOnClickListener {
            weekOffset--
            updateInsights()
        }
        findViewById<MaterialButton>(R.id.btnNextWeek).setOnClickListener {
            if (weekOffset < 0) { weekOffset++; updateInsights() }
        }
        updateInsights()
    }

    override fun onSupportNavigateUp(): Boolean { finish(); return true }

    private fun updateInsights() {
        val (weekDates, weekLabel) = getWeekDates(weekOffset)
        val prevWeekDates = getWeekDates(weekOffset - 1).first

        val weekCounts = weekDates.map { Prefs.getCbdCount(this, it) + Prefs.getThcCount(this, it) }
        val prevCounts = prevWeekDates.map { Prefs.getCbdCount(this, it) + Prefs.getThcCount(this, it) }

        val weekAvg = weekCounts.average()
        val prevAvg = prevCounts.average()

        // Semana label
        findViewById<TextView>(R.id.tvWeekLabel).text = weekLabel

        // Botón siguiente: deshabilitado si es semana actual
        findViewById<MaterialButton>(R.id.btnNextWeek).isEnabled = weekOffset < 0

        // Barras
        renderBars(weekCounts, weekDates)

        // Comparativa
        val avgText = "Promedio: %.1f CBD+THC/día".format(weekAvg)
        findViewById<TextView>(R.id.tvAvgLabel).text = avgText

        val compareText = if (prevAvg > 0) {
            val pct = ((weekAvg - prevAvg) / prevAvg * 100).roundToInt()
            val arrow = if (pct >= 0) "▲" else "▼"
            val color = if (pct > 0) "#e76f51" else "#52b788"
            "$arrow ${abs(pct)}% vs semana anterior"
        } else ""
        findViewById<TextView>(R.id.tvCompareLabel).text = compareText

        // Tip
        updateTip()
    }

    private fun renderBars(counts: List<Int>, dates: List<String>) {
        val container = findViewById<LinearLayout>(R.id.llBarsContainer)
        val labelsContainer = findViewById<LinearLayout>(R.id.llDayLabels)
        container.removeAllViews()
        labelsContainer.removeAllViews()

        val maxCount = counts.maxOrNull()?.takeIf { it > 0 } ?: 1
        val dayNames = listOf("L", "M", "X", "J", "V", "S", "D")
        val containerHeight = resources.displayMetrics.density * 140

        counts.forEachIndexed { i, count ->
            val heightFraction = count.toFloat() / maxCount
            val barHeight = (containerHeight * heightFraction).toInt().coerceAtLeast(4)

            val wrapper = LinearLayout(this).apply {
                orientation = LinearLayout.VERTICAL
                gravity = Gravity.BOTTOM or Gravity.CENTER_HORIZONTAL
                layoutParams = LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.MATCH_PARENT, 1f)
            }

            val bar = View(this).apply {
                layoutParams = LinearLayout.LayoutParams(
                    LinearLayout.LayoutParams.MATCH_PARENT, barHeight
                ).also { it.setMargins(4, 0, 4, 0) }
                setBackgroundColor(HeatmapUtils.colorForCount(count))
                background?.also {
                    val r = resources.displayMetrics.density * 4
                    background = android.graphics.drawable.GradientDrawable().apply {
                        setColor(HeatmapUtils.colorForCount(count))
                        cornerRadii = floatArrayOf(r, r, r, r, 0f, 0f, 0f, 0f)
                    }
                }
            }
            wrapper.addView(bar)
            container.addView(wrapper)

            // Label día
            val label = TextView(this).apply {
                text = dayNames[i]
                textAlignment = View.TEXT_ALIGNMENT_CENTER
                layoutParams = LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1f)
                setTextColor(Color.GRAY)
                textSize = 12f
            }
            labelsContainer.addView(label)
        }
    }

    private fun updateTip() {
        val cardTip = findViewById<CardView>(R.id.cardTip)
        val tvTip = findViewById<TextView>(R.id.tvTip)

        val dayAvgs = (0..6).map { dayOfWeek ->
            val allDates = getAllDatesWithData()
            val cal = Calendar.getInstance()
            val totals = allDates.mapNotNull { dateKey ->
                runCatching { fmt.parse(dateKey) }.getOrNull()?.let { date ->
                    cal.time = date
                    val dow = (cal.get(Calendar.DAY_OF_WEEK) - Calendar.MONDAY + 7) % 7
                    if (dow == dayOfWeek) Prefs.getCbdCount(this, dateKey) + Prefs.getThcCount(this, dateKey)
                    else null
                }
            }
            totals.takeIf { it.isNotEmpty() }?.average() ?: 0.0
        }

        val globalAvg = dayAvgs.filter { it > 0 }.average().takeIf { !it.isNaN() } ?: return
        val maxDayIdx = dayAvgs.indices.maxByOrNull { dayAvgs[it] } ?: return
        val maxDayAvg = dayAvgs[maxDayIdx]
        if (maxDayAvg <= 0 || globalAvg <= 0) { cardTip.visibility = View.GONE; return }

        val weeks = getAllDatesWithData().size / 7
        if (weeks < 2) { cardTip.visibility = View.GONE; return }

        val pct = ((maxDayAvg - globalAvg) / globalAvg * 100).roundToInt()
        if (pct < 15) { cardTip.visibility = View.GONE; return }

        val dayNames = listOf("lunes", "martes", "miércoles", "jueves", "viernes", "sábados", "domingos")
        tvTip.text = "💡 Los ${dayNames[maxDayIdx]} consumes un $pct% más de lo habitual."
        cardTip.visibility = View.VISIBLE
    }

    private fun getAllDatesWithData(): List<String> {
        val sp = getSharedPreferences("CBDCounter", MODE_PRIVATE)
        return sp.all.keys
            .filter { it.startsWith("count_cbd_") }
            .map { it.removePrefix("count_cbd_") }
    }

    private fun getWeekDates(offset: Int): Pair<List<String>, String> {
        val cal = Calendar.getInstance()
        val dow = (cal.get(Calendar.DAY_OF_WEEK) - Calendar.MONDAY + 7) % 7
        cal.add(Calendar.DAY_OF_YEAR, -dow + (offset * 7))

        val dates = (0..6).map { i ->
            val d = cal.clone() as Calendar
            d.add(Calendar.DAY_OF_YEAR, i)
            fmt.format(d.time)
        }

        val label = if (offset == 0) "Esta semana" else {
            val start = SimpleDateFormat("dd/MM", Locale.getDefault()).format(cal.time)
            val end = (cal.clone() as Calendar).also { it.add(Calendar.DAY_OF_YEAR, 6) }
            "$start – ${SimpleDateFormat("dd/MM", Locale.getDefault()).format(end.time)}"
        }
        return Pair(dates, label)
    }
}
```
